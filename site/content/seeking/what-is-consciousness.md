<div class="insight-text">Consciousness is known and experienced through the functioning of the mind and body, because it is only by the operations of the mind that we are able to recognise consciousness at all.</div>

<div class="insight-label">Fifth aspect</div>
<div class="insight-text">Without the mind and body, consciousness is still there and it does not cease to exist, but it cannot be known or experienced. </div>
<div class="analogy-box">

(function(){
var labels = ['Introduction', 'I. Not a part, not a product', 'II. Pervades and illumines', 'III. Not limited by', 'IV. Known through reflection', 'V. Still there, not experienced'];
var current = 0;
var total = 6;
var dotsEl = document.getElementById('cw-dots');
var scenes = document.querySelectorAll('.cw .scene');
for (var i = 0; i < total; i++) {
(function(idx){
var d = document.createElement('div');
d.className = 'dot' + (idx === 0 ? ' active' : '');
d.onclick = function(){ cwGoTo(idx) };
dotsEl.appendChild(d);
})(i);
}
function cwGoTo(n) {
if (n < 0 || n >= total) return;
scenes[current].classList.remove('active');
dotsEl.children[current].classList.remove('active');
dotsEl.children[current].classList.add('visited');
current = n;
scenes[current].classList.add('active');
dotsEl.children[current].classList.remove('visited');
dotsEl.children[current].classList.add('active');
document.getElementById('cw-prev').disabled = current === 0;
document.getElementById('cw-next').disabled = current === total - 1;
document.getElementById('cw-step').textContent = labels[current];
}
window.cwGo = function(dir) { cwGoTo(current + dir); };
})();
