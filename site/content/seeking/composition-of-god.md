(function(){
var labels = ['The Structure', 'I. Physical Body', 'II. Subtle Body', 'III. Causal Body', 'IV. Consciousness', 'V. The Complete Map'];
var current = 0;
var total = 6;
var dotsEl = document.getElementById('cog-dots');
var scenes = document.querySelectorAll('.cog .scene');
for (var i = 0; i < total; i++) {
(function(idx){
var d = document.createElement('div');
d.className = 'dot' + (idx === 0 ? ' active' : '');
d.onclick = function(){ cogGoTo(idx) };
dotsEl.appendChild(d);
})(i);
}
function cogGoTo(n) {
if (n < 0 || n >= total) return;
scenes[current].classList.remove('active');
dotsEl.children[current].classList.remove('active');
dotsEl.children[current].classList.add('visited');
current = n;
scenes[current].classList.add('active');
dotsEl.children[current].classList.remove('visited');
dotsEl.children[current].classList.add('active');
document.getElementById('cog-prev').disabled = current === 0;
document.getElementById('cog-next').disabled = current === total - 1;
document.getElementById('cog-step').textContent = labels[current];
}
window.cogGo = function(dir) { cogGoTo(current + dir); };
})();
